\begin{abstract}
In \emph{From a Failed Corner Slide to a Verified Tableau Injection} (\texttt{main.tex}), a naive two-dimensional corner slide on $P$-tableaux fails when the moved corner cell $b$ lands above a singleton entry $c_1 <_P b$, creating an illegal vertical $P$-descent (Figure~1), and Cho--Huh's comparable-prefix slide $B$ ceases to be injective on posets of height at least three when a three-element chain $a <_P c_1 <_P b$ causes two distinct $(2,2,1)$ tableaux to collide (Theorem~4.3).

This article presents the resolution of both obstructions---together with a simultaneous construction for the $a \ge 3$ sectors of the Kanade--Russell $I_1$ (KR1) partition bijection ($\mathcal{B} \leftrightarrow \mathcal{U} \leftrightarrow \mathcal{C} \leftrightarrow \mathcal{A}$)---originating from \textbf{Christopher Yamba's geometric insight and suggestion}: instead of restricting slides to a two-dimensional Ferrers diagram $(x, y)$, slide the obstructed corner cell into a three-dimensional Ferrers diagram $(x, y, z)$ where entries strictly descend in $P$ along the extra third axis, and extend the configuration space to $d > 3$ dimensions when higher-order chains or multi-scale column weights interact. All core combinatorial and algebraic theorems are accompanied by a machine-checked Lean~4.34.0 (\texttt{Std}-only) supplementary certificate (\texttt{MultidimensionalFerrers.lean}) and exact Python verification suites.
\end{abstract}

\section{Introduction and Attribution of the Geometric Insight}

Let $P = ([n], <_P)$ be a finite poset and $G = \mathrm{inc}(P)$ its incomparability graph. In Shareshian--Wachs and Gasharov's tableau expansion of the chromatic quasisymmetric function $X_G(x; q)$, a $P$-tableau $T \in \mathcal{T}_P(\mu)$ of shape $\mu \vdash n$ satisfies two planar order conditions:
\begin{enumerate}
  \item \textbf{Row-strict $P$-chains ($x$-axis):} $T_{i,j} <_P T_{i,j+1}$ along each row.
  \item \textbf{Column non-descent ($y$-axis):} $T_{i+1,j} \not<_P T_{i,j}$ down each column.
\end{enumerate}

When $T$ has shape $(2^{k+1}, 1^m)$ with last two-cell row $(a, b)$ followed by singleton leg entries $c_1, \dots, c_m$, the naive two-dimensional slide splits $(a, b)$ into $(a), (b)$ above $(c_1)$. Whenever $c_1 <_P b$, this creates a vertical descent $c_1 <_P b$ along the $y$-axis (Figure~1 of \texttt{main.tex}).

\begin{remark}[Christopher Yamba's $3\mathrm{D}$ and $d\mathrm{D}$ Ferrers Principle]
Inspecting Figure~1 of \texttt{main.tex}, Christopher Yamba observed that if one embeds the tableau in three-dimensional space $(x, y, z)$ and slides the corner cell $b$ along the \textbf{third axis ($z$-axis)} where entries strictly descend in $P$ ($b >_P c_1 >_P \dots >_P c_p$), the planar column obstruction disappears; furthermore, extending from $3\mathrm{D}$ to $d > 3$ dimensions provides a natural coordinate system for multi-sector cancellations and multi-scale partition bijections.
\end{remark}

\section{3D Fiber Dichotomy and Resolution of the Height-3 Collision}

\begin{theorem}[Non-3-Chain Local Fiber Injectivity; \texttt{bubble\_injective\_local\_fiber}]
\label{thm:local-fiber}
Let $r$ be a decidable relation on $\mathbb{N}$, and define the local non-three-chain predicate
\[
  \mathrm{NoLocalThreeChain}(r, a, b) \;:=\; \forall z,\; r(z, b) \implies \neg\, r(a, z).
\]
Then for any $b, b'$ and singleton tails $\mathit{tail}, \mathit{tail}'$ satisfying $r(a, b)$, $r(a, b')$, $\mathrm{NoLocalThreeChain}(r, a, b)$, and $\mathrm{NoLocalThreeChain}(r, a, b')$, equality of Cho--Huh's comparable-prefix insertions
\[
  \mathrm{bubbleInsert}(r, b, \mathit{tail}) = \mathrm{bubbleInsert}(r, b', \mathit{tail}')
\]
implies $b = b'$ and $\mathit{tail} = \mathit{tail}'$, without any global height restriction on $r$.
\end{theorem}

When $\mathrm{NoLocalThreeChain}(r, a, b)$ fails on $\mathcal{T}_P(2,2,1)$---that is, when $T = ((u,v),(a,b),(c))$ satisfies $a <_P c <_P b$ and $c \not<_P v$---its image $B(T) = ((u,v),(a),(c),(b))$ collides with $B(T')$ for $T' = ((u,v),(a,c),(b)) \in \mathcal{T}_P(2,2,1)$. Rotating the 3-chain along the third Ferrers axis into shape $(3,2)$ resolves the collision:

\begin{theorem}[3D Rotation into Shape $(3,2)$; \texttt{promote32\_tableau\_valid}, \texttt{promote32\_tableau\_graph\_inversions}]
\label{thm:promote32}
Let $r$ be a transitive relation on $\mathbb{N}$ and $T = [[u,v],[a,b],[c]]$ a valid tableau satisfying $r(a, c)$, $r(v, b)$, and $\neg\, r(c, v)$. Then the 3D-rotated filling
\[
  \mathrm{promote32}(u, v, a, b, c) := [[u, v, b], [a, c]]
\]
is a valid tableau of shape $(3,2)$ on the same vertex set and preserves the row-based graph inversion statistic:
\[
  \mathrm{tableauGraphInversions}(r, \mathrm{promote32}(u, v, a, b, c))
  = \mathrm{tableauGraphInversions}(r, [[u, v], [a, b], [c]]).
\]
\end{theorem}

\section{Universal Interior Hook Slides and 6-Sector Cancellation}

\begin{theorem}[Universal Endpoint and Interior Hook Slides; \texttt{hookBubbleEnd\_*}, \texttt{hookBubbleMid\_*}]
\label{thm:hook-slides}
Let $r$ be an asymmetric decidable relation on $\mathbb{N}$ and let $T = [u, v, w] :: \mathrm{singletons}(\mathit{tail})$ be a valid hook tableau of shape $(3, 1^m)$. Then both the endpoint hook slide
\[
  \mathrm{hookBubbleEnd}(r, u, v, w, \mathit{tail}) := [u, v] :: \mathrm{singletons}(\mathrm{bubbleInsert}(r, w, \mathit{tail}))
\]
and the interior hook slide
\[
  \mathrm{hookBubbleMid}(r, u, v, w, \mathit{tail}) := [u, w] :: \mathrm{singletons}(\mathrm{bubbleInsert}(r, v, \mathit{tail}))
\]
are valid tableaux of shape $(2, 1^{m+1})$ with $\mathrm{tableauGraphInversions}$ equal to that of $T$.
\end{theorem}

\begin{theorem}[$n$-Dimensional $\beta$-Rearrangement Bijection for Dual Jacobi--Trudi Sectors]
\label{thm:beta-bijection}
Fix $n \ge 1$ and $\lambda = (\lambda_1, \dots, \lambda_k) \vdash n$, and let $\hat{\lambda} = (\lambda_1, \dots, \lambda_k, 0^{n-k}) \in \mathbb{N}^n$. The non-vanishing determinant terms $(\mu, \sigma)$ contributing to $[e_\lambda] \det_{1 \le i,j \le n}(e_{\mu'_i - i + j})$ are in canonical bijection with rearrangements $\beta = (\beta_0, \dots, \beta_{n-1}) \in \mathrm{Perm}(\hat{\lambda})$ whose shifted keys $\kappa_j(\beta) = \beta_j - j$ are pairwise distinct.
\end{theorem}

\section{Universal $k$-Dimensional First-Column Cascade in \texttt{SignedArrays}}

\begin{theorem}[Universal $(0, k) \leftrightarrow (k-1, 1)$ Cascade; \texttt{cascade\_split\_preserves\_all}]
\label{thm:cascade}
For every decidable relation $r$, every non-empty list $\mathit{as}$ such that $(b :: \mathit{as})$ is a strict $r$-chain, and every signed-array suffix $\mathit{tail}$, the adjacent-row toggle between $((p, []), (q, b :: \mathit{as}))$ and $((q, \mathit{as}), (p, [b]))$ preserves \texttt{ShapeAt}, \texttt{Good}, and \texttt{graphInversions} for all $k = |b :: \mathit{as}| \ge 2$.
\end{theorem}

\section{6D Ferrers Cones and the Kanade--Russell $I_1$ Bijection}

\begin{theorem}[Scale-3 Self-Similarity and 6D Octant Weight Identities; \texttt{kr1\_*}]
\label{thm:kr1-identities}
Let $Q(a, b) = a^2 + 3ab + 3b^2$ be the Kanade--Russell $I_1$ base quadratic form. For all $c, r, b \in \mathbb{N}$ and $a \ge 1$:
\begin{align}
  Q(3c + r, b) &= r^2 + 3r(b + 2c) + 3\,Q(b, c), \label{eq:self-sim}\\
  Q(3c, b) &= 3\,Q(b, c), \label{eq:scale3}\\
  Q(a, b) &= 3b(b + 1) + a^2 + 3(a - 1)b, \label{eq:head-offset}\\
  (9 + 2r_2) + 6q_2 + 3d_3 + d_1 &= Q(3, 0) + 3d_3 + 2(3q_2 + r_2) + d_1. \label{eq:octant}
\end{align}
\end{theorem}

\begin{theorem}[The 6D Ferrers Cone Theorem and Universal $(a=3, b=0)$ Bijection]
\label{thm:6d-cone}
In the 6-dimensional column multiplicity space $(10^z, 9^y, 8^x, 6^{c_6}, 3^{c_3}, 1^{c_1})$, every head $(x, y, z) \in \mathbb{N}^3$ with $y \in \{1, 2\}$ and $x + z \ge 1$ ($0 \le z \le 3$) spans a full 3D octant $(c_6, c_3, c_1) \in \mathbb{N}^3$ (with $c_6 + c_3 + c_1 \ge 1$ when $x = 0$) inside $\mathcal{U} \setminus \mathcal{U}_{a \le 2}$. Together with equation~\eqref{eq:octant}, this yields an explicit, lookup-free, weight-preserving bijection for the entire infinite sector $(a=3, b=0, \forall \mu)$ (verified across all 710 states through weight $N = 35$) and completes the round-trip bijection $\mathcal{B} \leftrightarrow \mathcal{U} \leftrightarrow \mathcal{C} \leftrightarrow \mathcal{A}$ across all 758 states for $0 \le N \le 26$.
\end{theorem}
